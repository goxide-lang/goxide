// Compare parsed AST structure/scalars, including token-presence fields, while
// ignoring only numeric positions and parser scope/object caches.
package main

import (
 "crypto/sha256"
 "encoding/json"
 "fmt"
 "go/parser"
 "go/token"
 "os"
 "reflect"
)
var position = reflect.TypeOf(token.NoPos)
func shape(v reflect.Value) any {
 if !v.IsValid(){return nil}
 if v.Type()==position{return v.Int()!=0}
 switch v.Kind(){
 case reflect.Interface,reflect.Pointer:
  if v.IsNil(){return nil};return shape(v.Elem())
 case reflect.Struct:
  m:=map[string]any{"$type":v.Type().String()}
  for i:=0;i<v.NumField();i++ {f:=v.Type().Field(i);switch f.Name{case "Obj","Scope","Unresolved","Comments","Doc","Comment":continue};m[f.Name]=shape(v.Field(i))};return m
 case reflect.Slice:
  if v.IsNil(){return nil};a:=make([]any,v.Len());for i:=range a{a[i]=shape(v.Index(i))};return a
 case reflect.String:return v.String()
 case reflect.Bool:return v.Bool()
 case reflect.Int,reflect.Int8,reflect.Int16,reflect.Int32,reflect.Int64:return v.Int()
 case reflect.Uint,reflect.Uint8,reflect.Uint16,reflect.Uint32,reflect.Uint64:return v.Uint()
 default:panic(v.Type().String())
 }
}
func digest(path string) (string,error){
 f,e:=parser.ParseFile(token.NewFileSet(),path,nil,parser.SkipObjectResolution);if e!=nil{return "",e}
 b,e:=json.Marshal(shape(reflect.ValueOf(f)));if e!=nil{return "",e};return fmt.Sprintf("%x",sha256.Sum256(b)),nil
}
func main(){
 if len(os.Args)!=3{panic("two Go files required")}
 a,e:=digest(os.Args[1]);if e!=nil{panic(e)};b,e:=digest(os.Args[2]);if e!=nil{panic(e)}
 json.NewEncoder(os.Stdout).Encode(map[string]any{"baseline":os.Args[1],"current":os.Args[2],"baseline_shape_sha256":a,"current_shape_sha256":b,"identical":a==b})
 if a!=b{os.Exit(1)}
}
